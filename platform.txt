# arduino-stc51 plain-C core and platform.
# See docs/toolchain-and-sdk.md for compiler provenance and support boundaries.

name=arduino-stc51
version=0.0.1

compiler.warning_flags=--less-pedantic
compiler.warning_flags.none=--less-pedantic
compiler.warning_flags.default=--less-pedantic
compiler.warning_flags.more=--less-pedantic -Wp-Wall
compiler.warning_flags.all=--less-pedantic -Wp-Wall -V

compiler.path={runtime.tools.sdcc-mcs251.path}/bin
compiler.shell.cmd=sh
compiler.shell.cmd.macosx=/bin/sh
compiler.shell.cmd.windows={runtime.tools.MCS51Tools.path}/win/busybox.exe
compiler.shell.args=
compiler.shell.args.windows=ash
compiler.wrapper.path={runtime.platform.path}/tools/wrapper

compiler.c.cmd=sdcc
compiler.c.wrapper=sdcc.sh
compiler.dep.flags=-MMD
compiler.c.flags=-c --std-sdcc11 --opt-code-size -Ddouble=float --model-{build.memory_model} {compiler.warning_flags}
compiler.c.elf.cmd=sdcc
compiler.c.elf.wrapper=sdcc-link.sh
compiler.c.elf.flags=--code-size {upload.maximum_size} --xram-size {upload.maximum_xdata_size} --model-{build.memory_model}

# Arduino sketches use a .cpp extension, but this is intentionally a plain-C
# core. MCS51Tools converts the generated sketch to C before invoking SDCC.
compiler.cpp.cmd=sdcc
compiler.cpp.wrapper=sdcc.sh
compiler.cpp.flags={compiler.c.flags}

compiler.S.flags=-c -g -x assembler-with-cpp
compiler.ar.path={runtime.tools.sdcc-mcs251.path}/bin
compiler.ar.path.windows={runtime.tools.MCS51ArchiveTools.path}/bin
compiler.ar.cmd=sdar
compiler.ar.wrapper=sdar.sh
compiler.ar.flags=rcs
compiler.systemincludes="-I{runtime.tools.sdcc-mcs251.path}/include" "-I{runtime.tools.sdcc-mcs251.path}/include/mcs51"

build.extra_flags=
compiler.c.extra_flags=
compiler.c.elf.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
compiler.ar.extra_flags=

recipe.c.o.pattern="{compiler.shell.cmd}" {compiler.shell.args} "{compiler.wrapper.path}/stc-compile.sh" "{compiler.path}/{compiler.c.cmd}" "{source_file}" "{object_file}" re1 {compiler.dep.flags} {compiler.c.flags} {build.target} -D{build.mcu_macro} -DF_CPU={build.f_cpu} {build.mode_flags} {build.core_flags} {build.timer_flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.c.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}

recipe.cpp.o.pattern="{compiler.shell.cmd}" {compiler.shell.args} "{compiler.wrapper.path}/stc-compile.sh" "{compiler.path}/{compiler.cpp.cmd}" "{source_file}" "{object_file}" re2 {compiler.dep.flags} {compiler.cpp.flags} {build.target} -D{build.mcu_macro} -DF_CPU={build.f_cpu} {build.mode_flags} {build.core_flags} {build.timer_flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.cpp.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}

recipe.S.o.pattern="{compiler.path}/{compiler.c.cmd}" {compiler.S.flags} {build.target} -D{build.mcu_macro} -DF_CPU={build.f_cpu} {build.mode_flags} {build.core_flags} {build.timer_flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.S.extra_flags} {build.extra_flags} {includes} "{source_file}" -o "{object_file}"

# The MCS251 release's sdar is missing runtime DLLs. A separately pinned,
# working SDCC archive helper creates core.a; compilation and linking still use
# SDCC 4.6.0 / MCS251 exclusively.
recipe.ar.pattern="{compiler.shell.cmd}" {compiler.shell.args} "{compiler.wrapper.path}/stc-archive.sh" "{compiler.ar.path}/{compiler.ar.cmd}" "{archive_file_path}" "{object_file}" {compiler.ar.flags} {compiler.ar.extra_flags}

recipe.c.combine.pattern="{compiler.shell.cmd}" {compiler.shell.args} "{compiler.wrapper.path}/stc-link.sh" "{compiler.path}/{compiler.c.elf.cmd}" {build.target} {compiler.c.elf.flags} {compiler.c.elf.extra_flags} {object_files} "{build.path}/{archive_file}" --out-fmt-ihx -o "{build.path}/{build.project_name}.hex"

recipe.output.tmp_file={build.project_name}.hex
recipe.output.save_file={build.project_name}.{build.variant}.hex

# SDCC writes a human-readable .mem file next to the Intel HEX output. The
# wrapper includes internal DATA/IDATA usage as well as PDATA and XDATA.
recipe.size.pattern="{compiler.shell.cmd}" {compiler.shell.args} "{compiler.wrapper.path}/stc-size.sh" "{build.path}/{build.project_name}.mem"
recipe.size.regex=^STC_PROGRAM_BYTES ([0-9]+)$
recipe.size.regex.data=^STC_RAM_BYTES ([0-9]+)$

# Arduino CLI/IDE library discovery and prototype generation.
preproc.includes.flags=-M -MG -MP
recipe.preproc.includes="{compiler.shell.cmd}" {compiler.shell.args} "{compiler.wrapper.path}/stc-compile.sh" "{compiler.path}/{compiler.cpp.cmd}" "{source_file}" "{preprocessed_file_path}" re11 {preproc.includes.flags} {build.target} -D{build.mcu_macro} -DF_CPU={build.f_cpu} {build.mode_flags} {build.core_flags} {build.timer_flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.cpp.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}
preproc.macros.flags=-E -dM
recipe.preproc.macros="{compiler.shell.cmd}" {compiler.shell.args} "{compiler.wrapper.path}/stc-compile.sh" "{compiler.path}/{compiler.cpp.cmd}" "{source_file}" "{preprocessed_file_path}" re12 {preproc.macros.flags} {build.target} -D{build.mcu_macro} -DF_CPU={build.f_cpu} {build.mode_flags} {build.core_flags} {build.timer_flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.cpp.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}

# STC's current AiCube-ISP is intentionally not bundled or automated. Export
# the compiled HEX and program it with the verified version in sdk/manifest.json.
